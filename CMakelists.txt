cmake_minimum_required(VERSION 3.10)
project(sgl C CXX)

# 定义可执行文件目标
# add_executable(myapp demo/main.c)

# 设置C标准
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)

# 定义输出目录
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/output)
set(CMAKE_LIBRARY_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/output/lib)
set(CMAKE_ARCHIVE_OUTPUT_DIRECTORY ${CMAKE_BINARY_DIR}/output/lib)

# 主目录
set(SGL_ROOT_DIR ${CMAKE_CURRENT_SOURCE_DIR})
set(DEMO_DIR ${SGL_ROOT_DIR}/demo)

# SGL 库源文件
file(GLOB SGL_CORE_SOURCES ${SGL_ROOT_DIR}/sgl/source/core/*.c)
file(GLOB SGL_TIMER_SOURCES ${SGL_ROOT_DIR}/sgl/source/components/timer/*.c)
file(GLOB SGL_DRAW_SOURCES ${SGL_ROOT_DIR}/sgl/source/draw/*.c)
file(GLOB SGL_FONTS_SOURCES ${SGL_ROOT_DIR}/sgl/source/fonts/*.c)
file(GLOB SGL_MM_SOURCES ${SGL_ROOT_DIR}/sgl/source/mm/lwmem/*.c)
file(GLOB SGL_WIDGETS_SOURCES ${SGL_ROOT_DIR}/sgl/source/widgets/*/*.c)
file(GLOB SGL_FS_SOURCES ${SGL_ROOT_DIR}/sgl/source/fs/*/*.c)
# 排除误放入 widget 目录中的旧实现 / 参考文件（不可编译）
list(REMOVE_ITEM SGL_WIDGETS_SOURCES ${SGL_ROOT_DIR}/sgl/source/widgets/msgbox/lv_msgbox.c)
list(REMOVE_ITEM SGL_WIDGETS_SOURCES ${SGL_ROOT_DIR}/sgl/source/widgets/gif/sgl_gifdec.c)
# 排除 littlefs 官方参考代码（仅作参考，不参与编译，实现见 sgl_littlefs.c）
list(REMOVE_ITEM SGL_FS_SOURCES ${SGL_ROOT_DIR}/sgl/source/fs/littlefs/lfs.c)

# 合并所有 SGL 源文件
set(SGL_SOURCES
    ${SGL_CORE_SOURCES}
    ${SGL_TIMER_SOURCES}
    ${SGL_DRAW_SOURCES}
    ${SGL_FONTS_SOURCES}
    ${SGL_MM_SOURCES}
    ${SGL_WIDGETS_SOURCES}
    ${SGL_FS_SOURCES}
)

# 包含目录
include_directories(
    ${SGL_ROOT_DIR}/sgl/source
    ${SGL_ROOT_DIR}/sgl/source/include
    ${DEMO_DIR}/sdl/include/SDL2
)

# 创建SGL库
add_library(sgl STATIC ${SGL_SOURCES})

# 示例清单集中在 sgl/examples/build.cmake 里维护，这里只选要编译哪些示例：
#   sgl_examples_select(SGL_EXAMPLES)                      -> 全部示例
#   sgl_examples_select(SGL_EXAMPLES win filebrowser ...)   -> 只编译列出的示例
include(${SGL_ROOT_DIR}/sgl/examples/build.cmake)
sgl_examples_select(SGL_EXAMPLES)

# 模拟器本体 + 性能基准组件 + 上一步选出的示例
set(DEMO_SOURCES
    ${DEMO_DIR}/main.c
    ${DEMO_DIR}/sgl_port_sdl2.c
    ${SGL_ROOT_DIR}/sgl/source/components/bench/sgl_bench.c
    ${SGL_EXAMPLES}
)

set(SDL2_LIB_NAME
    mingw32 SDL2main SDL2
)
link_directories(
    ${DEMO_DIR}/sdl/lib
)

# 创建演示应用程序
add_executable(sgl_simulator ${DEMO_SOURCES})
target_compile_definitions(sgl_simulator PRIVATE
    SGL_AVI_DEMO_BASE_DIR=\"${SGL_ROOT_DIR}\"
)

# 链接库
target_link_libraries(sgl_simulator sgl ${SDL2_LIB_NAME})

add_custom_command(
    TARGET sgl_simulator POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy
        "${DEMO_DIR}/sgl_config.h"
        "${SGL_ROOT_DIR}/sgl/source/sgl_config.h"
    COMMENT "Force copying sgl_config.h (always overwrite)"
)

add_custom_command(
    TARGET sgl_simulator POST_BUILD
    COMMAND ${CMAKE_COMMAND} -E copy_if_different
        "${DEMO_DIR}/sdl/bin/SDL2.dll"
        $<TARGET_FILE_DIR:sgl_simulator>
    COMMENT "Copying SDL2.dll to output directory"
)

# 安装规则
install(TARGETS sgl_simulator
    RUNTIME DESTINATION bin
)


# 输出信息
message(STATUS "SGL project configured successfully")
message(STATUS "Build type: ${CMAKE_BUILD_TYPE}")
message(STATUS "Output directory: ${CMAKE_RUNTIME_OUTPUT_DIRECTORY}")
